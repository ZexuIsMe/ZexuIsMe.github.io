```text
  //编辑员工信息
  async editEmployee(item: Employee) {
    let pre = new relationalStore.RdbPredicates("employees")
    pre.equalTo("id", item.id)
    return this.rdbStore.update({
      name: item.name,
      salary: item.salary,
      dept: item.dept
    }, pre)
  }
```
