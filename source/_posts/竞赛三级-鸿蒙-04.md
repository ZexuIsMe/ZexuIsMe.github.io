## Step 2: 收集参数

```text
# 先注册一个参数
@state employee2: Employee = {id: 0, name: "", salary: 0, dept: ""}

# 绑定事件
TextInput({ placeholder: "员工名" })
  .width("80%").margin({top: 10})
  .onChange(val => {
    this.employee2.name = val      
  })
## 注意：薪资需要通过 Number(val) 转数字
```

## Step 3： 新增员工信息

```text
  //新增员工信息
  async addEmployee(employee2Add:Employee){
    return this.rdbStore.insert("employees", {
      name:employee2Add.name, salary:employee2Add.salary, dept:employee2Add.dept
    })
  }
```
